cmake_minimum_required(VERSION 3.16)
project(NetTrace VERSION 1.0.0 LANGUAGES CXX)
option(NETTRACE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(nettrace_core src/protocol.cpp src/socket.cpp src/log.cpp src/analyzer.cpp)
target_include_directories(nettrace_core PUBLIC include)
if(MSVC)
  target_compile_options(nettrace_core PUBLIC /W4 /WX)
else()
  target_compile_options(nettrace_core PUBLIC -Wall -Wextra -Wpedantic -Werror)
endif()
if(WIN32)
  target_link_libraries(nettrace_core PUBLIC ws2_32)
endif()
if(NETTRACE_SANITIZERS AND NOT MSVC AND NOT WIN32)
  target_compile_options(nettrace_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(nettrace_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(nettrace src/main.cpp src/client/client.cpp src/server/server.cpp)
find_package(Threads REQUIRED)
target_link_libraries(nettrace PRIVATE nettrace_core Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  add_executable(nettrace_tests tests/unit.cpp)
  target_link_libraries(nettrace_tests PRIVATE nettrace_core)
  add_test(NAME unit COMMAND nettrace_tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:nettrace>)
  set_tests_properties(integration PROPERTIES TIMEOUT 60)
endif()
install(TARGETS nettrace RUNTIME DESTINATION bin)
